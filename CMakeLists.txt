cmake_minimum_required(VERSION 3.21)
project(TabletProRadial VERSION 0.3.9 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options("$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
option(TABLETPRO_BUILD_DESKTOP "Build the Qt desktop editor and overlay" ON)
include(CTest)

add_library(radial_core core/model.cpp core/gesture.cpp)
target_include_directories(radial_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(radial_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(radial_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(radial_core_tests tests/core_tests.cpp)
  target_link_libraries(radial_core_tests PRIVATE radial_core)
  add_test(NAME portable_core COMMAND radial_core_tests)
endif()

if(TABLETPRO_BUILD_DESKTOP)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Network Qml Quick QuickControls2 Multimedia ShaderTools)
  set(CMAKE_AUTOMOC ON)
  add_library(radial_studio
    app/ProfileCodec.cpp app/ProfileCodec.h
    app/MenuSession.cpp app/MenuSession.h
    app/StudioController.cpp app/StudioController.h
    app/AudioFeedback.cpp app/AudioFeedback.h
    platform/WindowPlatform.cpp platform/WindowPlatform.h
    platform/StylusInput.cpp platform/StylusInput.h
    platform/BackdropEffect.cpp platform/BackdropEffect.h
    ui/RadialSurface.cpp ui/RadialSurface.h)
  target_link_libraries(radial_studio PUBLIC radial_core Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia)
  if(WIN32)
    target_link_libraries(radial_studio PRIVATE user32 gdi32 dwmapi)
    if(MSVC)
      add_library(TabletProGlass SHARED platform/GlassComposition.cpp)
      target_link_libraries(TabletProGlass PRIVATE windowsapp d2d1 dwmapi coremessaging ole32)
      set_target_properties(TabletProGlass PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lib")
      add_dependencies(radial_studio TabletProGlass)
      install(TARGETS TabletProGlass RUNTIME DESTINATION bin/lib)
    elseif(NOT CMAKE_CROSSCOMPILING)
      find_program(TABLETPRO_POWERSHELL NAMES pwsh powershell REQUIRED)
      set(TABLETPRO_GLASS_DLL "${CMAKE_CURRENT_BINARY_DIR}/lib/TabletProGlass.dll")
      add_custom_command(OUTPUT "${TABLETPRO_GLASS_DLL}"
        COMMAND "${TABLETPRO_POWERSHELL}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-glass-composition.ps1" -OutputDirectory "${CMAKE_CURRENT_BINARY_DIR}/lib"
        DEPENDS platform/GlassComposition.cpp scripts/build-glass-composition.ps1 VERBATIM)
      add_custom_target(radial_glass_composition DEPENDS "${TABLETPRO_GLASS_DLL}")
      add_dependencies(radial_studio radial_glass_composition)
      install(FILES "${TABLETPRO_GLASS_DLL}" DESTINATION bin/lib)
    endif()
  endif()
  qt_add_executable(tabletpro_radial app/main.cpp)
  add_library(radial_desktop app/UpdateChecker.cpp app/UpdateChecker.h app/DesktopShell.cpp app/DesktopShell.h)
  qt_add_resources(radial_desktop radial_brand PREFIX "/icons" BASE assets FILES assets/radial.ico)
  qt_add_resources(radial_desktop radial_history PREFIX "/history" BASE docs FILES docs/update-history.json)
  target_link_libraries(radial_desktop PUBLIC radial_studio Qt6::Widgets Qt6::Network)
  target_link_libraries(tabletpro_radial PRIVATE radial_desktop)
  target_compile_definitions(tabletpro_radial PRIVATE TABLETPRO_VERSION="${PROJECT_VERSION}")
  qt_add_resources(tabletpro_radial qml_source PREFIX "/" FILES qml/Main.qml qml/Overlay.qml qml/HubButton.qml qml/BloomLayer.qml)
  qt_add_shaders(tabletpro_radial bloom_shader PREFIX "/" FILES shaders/gentle-bloom.frag)
  set_target_properties(tabletpro_radial PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE)
  if(WIN32)
    enable_language(RC)
    configure_file(app/windows.rc.in "${CMAKE_CURRENT_BINARY_DIR}/radial-windows.rc" @ONLY)
    target_sources(tabletpro_radial PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/radial-windows.rc")
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/radial-windows.rc" PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/radial.ico")
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
      set(TABLETPRO_HID_DLL "${CMAKE_CURRENT_SOURCE_DIR}/vendor/tabletpro/TabletProHIDx64.dll")
    else()
      set(TABLETPRO_HID_DLL "${CMAKE_CURRENT_SOURCE_DIR}/vendor/tabletpro/TabletProHIDx32.dll")
    endif()
    add_custom_command(TARGET tabletpro_radial POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:tabletpro_radial>/lib"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${TABLETPRO_HID_DLL}" "$<TARGET_FILE_DIR:tabletpro_radial>/lib")
    install(FILES "${TABLETPRO_HID_DLL}" DESTINATION bin/lib)
  endif()
  if(BUILD_TESTING)
    add_executable(radial_live_update_check EXCLUDE_FROM_ALL tests/live_update_check.cpp)
    target_link_libraries(radial_live_update_check PRIVATE radial_desktop)
    add_executable(radial_update_tests tests/update_tests.cpp)
    target_link_libraries(radial_update_tests PRIVATE radial_desktop)
    target_compile_definitions(radial_update_tests PRIVATE TABLETPRO_VERSION="${PROJECT_VERSION}")
    qt_add_resources(radial_update_tests test_qml PREFIX "/" FILES qml/Main.qml qml/Overlay.qml qml/HubButton.qml qml/BloomLayer.qml)
    qt_add_shaders(radial_update_tests test_bloom_shader PREFIX "/" FILES shaders/gentle-bloom.frag)
    add_test(NAME tray_and_updates COMMAND radial_update_tests)
    set_tests_properties(tray_and_updates PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    add_executable(radial_backdrop_tests tests/backdrop_tests.cpp)
    target_link_libraries(radial_backdrop_tests PRIVATE radial_studio)
    add_test(NAME backdrop_lifecycle COMMAND radial_backdrop_tests)
    if(WIN32)
      add_executable(radial_settings_tests tests/settings_bloom_tests.cpp)
      target_link_libraries(radial_settings_tests PRIVATE radial_studio Qt6::Widgets)
      qt_add_resources(radial_settings_tests settings_test_qml PREFIX "/" FILES qml/Main.qml qml/Overlay.qml qml/HubButton.qml qml/BloomLayer.qml)
      qt_add_shaders(radial_settings_tests settings_test_shader PREFIX "/" FILES shaders/gentle-bloom.frag)
      add_test(NAME settings_bloom_rendered COMMAND radial_settings_tests)
      set_tests_properties(settings_bloom_rendered PROPERTIES RUN_SERIAL TRUE TIMEOUT 45)
      add_executable(radial_modifier_native_tests tests/modifier_native_tests.cpp)
      target_link_libraries(radial_modifier_native_tests PRIVATE radial_studio user32)
      add_test(NAME modifier_native_input COMMAND radial_modifier_native_tests)
      set_tests_properties(modifier_native_input PROPERTIES RUN_SERIAL TRUE TIMEOUT 30)
      add_executable(radial_glass_visual_tests tests/glass_visual_tests.cpp)
      target_link_libraries(radial_glass_visual_tests PRIVATE radial_studio)
      add_test(NAME glass_rendered_pixels COMMAND radial_glass_visual_tests)
      set_tests_properties(glass_rendered_pixels PROPERTIES RUN_SERIAL TRUE)
    endif()
    if(NOT WIN32)
      set_tests_properties(backdrop_lifecycle PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
    add_executable(radial_stylus_tests tests/stylus_input_tests.cpp)
    target_link_libraries(radial_stylus_tests PRIVATE radial_studio)
    add_test(NAME stylus_input COMMAND radial_stylus_tests)
    set_tests_properties(stylus_input PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
  endif()
  if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/app_tests.cpp")
    add_executable(radial_app_tests tests/app_tests.cpp)
    target_link_libraries(radial_app_tests PRIVATE radial_studio)
    add_test(NAME editor_and_format COMMAND radial_app_tests)
    set_tests_properties(editor_and_format PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
  endif()
  if(BUILD_TESTING AND NOT WIN32)
    add_executable(radial_platform_tests tests/platform_tests.cpp)
    target_link_libraries(radial_platform_tests PRIVATE radial_studio)
    add_test(NAME platform_simulation COMMAND radial_platform_tests)
    set_tests_properties(platform_simulation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
  endif()
  install(TARGETS tabletpro_radial BUNDLE DESTINATION . RUNTIME DESTINATION bin)
endif()
